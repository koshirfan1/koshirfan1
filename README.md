let name   = "Irfan Khan"
let builds = ["a programming language", "android tools", "python apps"]
let status = "building Kosh towards v1.0"

fn say_hi(who, things):
    print "Hi, I'm " + who
    for thing in things:
        print "  -> I build " + thing

say_hi(name, builds)
print "Status: " + status
